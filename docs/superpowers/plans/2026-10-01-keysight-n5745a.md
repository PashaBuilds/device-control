```ts
/**
 * Keysight N5745A (N5700 serisi, tek çıkış 30 V / 25 A) SCPI simülatörü.
 *
 * Gerçek cihazın LAN raw-socket (TCP 5025) komut setinin uygulama
 * kapsamındaki alt kümesini konuşur; kanal listesi yoktur (`VOLT 12`).
 * Yanıtlar LF ile biter. Aralıklar N5700 User's Guide'daki 30 V modeline
 * göredir: VOLT 0–31.5 (ve ≤ OVP/1.05), OVP 2–36 (ve ≥ VOLT×1.05).
 *
 * `SIM:` önekli komutlar yalnızca test içindir; gerçek sürücü göndermez.
 */

const V_MAX = 31.5
const I_MAX = 25
const OVP_MIN = 2
const OVP_MAX = 36

const ERR_UNDEF = -113 // Undefined header
const ERR_CONFLICT = -221 // Settings conflict
const ERR_RANGE = -222 // Data out of range

// Questionable Condition register bitleri
const QUES_OV = 1
const QUES_OC = 2

const BOOL = /^(ON|OFF|1|0)$/i

export class N5745aSim {
  vSet = 0
  iSet = 0
  ovp = OVP_MAX
  ocpEnabled = false
  outputOn = false
  ovTripped = false
  ocTripped = false
  /** Trip anındaki çıkış durumu; OUTP:PROT:CLE bunu geri yükler. */
  private outputBeforeTrip = false
  loadOhms = 10
  private errors: number[] = []

  reset(): void {
    const load = this.loadOhms
    Object.assign(this, new N5745aSim())
    this.loadOhms = load
  }

  private get tripped(): boolean {
    return this.ovTripped || this.ocTripped
  }

  private trip(kind: 'ov' | 'oc'): void {
    if (!this.tripped) this.outputBeforeTrip = this.outputOn
    if (kind === 'ov') this.ovTripped = true
    else this.ocTripped = true
    this.outputOn = false
  }

  /** Anlık ölçüm; gerekirse OVP/OCP tripler. */
  private measure(): { v: number; i: number } {
    if (!this.outputOn || this.tripped) return { v: 0, i: 0 }
    let v = this.vSet
    let i = this.loadOhms > 0 ? v / this.loadOhms : Number.POSITIVE_INFINITY
    let cc = false
    if (i > this.iSet) {
      cc = true
      i = this.iSet
      v = this.iSet * this.loadOhms
    }
    if (v > this.ovp) {
      this.trip('ov')
      return { v: 0, i: 0 }
    }
    if (cc && this.ocpEnabled) {
      this.trip('oc')
      return { v: 0, i: 0 }
    }
    return { v, i }
  }

  private fmt(x: number): string {
    return x.toExponential(5).toUpperCase() // ör. "1.20000E+1"
  }

  private set(value: string, min: number, max: number, apply: (x: number) => void): void {
    const x = Number(value)
    if (!Number.isFinite(x) || x < min - 1e-9 || x > max + 1e-9) {
      this.errors.push(ERR_RANGE)
      return
    }
    apply(x)
  }

  handle(rawLine: string): string | null {
    // İsteğe bağlı SOURce: önekini at
    const line = rawLine.trim().replace(/^SOUR(?:ce)?:/i, '')
    if (line === '') return null
    let m: RegExpMatchArray | null

    // --- Test uzantıları ---
    if ((m = line.match(/^SIM:LOAD\s+([\d.]+)$/i))) {
      this.loadOhms = Number(m[1])
      return null
    }
    if ((m = line.match(/^SIM:TRIP\s+(OVP|OCP)$/i))) {
      this.trip(m[1].toUpperCase() === 'OVP' ? 'ov' : 'oc')
      return null
    }

    // --- Ortak ---
    if (/^\*IDN\?$/i.test(line)) return 'Agilent Technologies,N5745A,US17M1234P,A.03.02,A.01.01'
    if (/^\*RST$/i.test(line)) {
      this.reset()
      return null
    }
    if (/^\*CLS$/i.test(line)) {
      this.errors = []
      return null
    }
    if (/^\*OPC\?$/i.test(line)) return '1'
    if (/^SYST(?:em)?:ERR(?:or)?\?$/i.test(line)) {
      const code = this.errors.shift()
      if (code === undefined) return '+0,"No error"'
      const msg =
        code === ERR_RANGE ? 'Data out of range' : code === ERR_CONFLICT ? 'Settings conflict' : 'Undefined header'
      return `${code},"${msg}"`
    }

    // --- Sorgular ---
    if (/^VOLT(?:age)?\?$/i.test(line)) return this.fmt(this.vSet)
    if (/^CURR(?:ent)?\?$/i.test(line)) return this.fmt(this.iSet)
    if (/^VOLT(?:age)?:PROT(?:ection)?(?::LEV(?:el)?)?\?$/i.test(line)) return this.fmt(this.ovp)
    if (/^CURR(?:ent)?:PROT(?:ection)?:STAT(?:e)?\?$/i.test(line)) return this.ocpEnabled ? '1' : '0'
    if (/^OUTP(?:ut)?(?::STAT(?:e)?)?\?$/i.test(line)) {
      this.measure()
      return this.outputOn ? '1' : '0'
    }
    if (/^MEAS(?:ure)?(?::SCAL(?:ar)?)?:VOLT(?:age)?(?::DC)?\?$/i.test(line)) return this.fmt(this.measure().v)
    if (/^MEAS(?:ure)?(?::SCAL(?:ar)?)?:CURR(?:ent)?(?::DC)?\?$/i.test(line)) return this.fmt(this.measure().i)
    if (/^STAT(?:us)?:QUES(?:tionable)?:COND(?:ition)?\?$/i.test(line)) {
      this.measure()
      return String((this.ovTripped ? QUES_OV : 0) | (this.ocTripped ? QUES_OC : 0))
    }

    // --- Ayarlar ---
    if ((m = line.match(/^VOLT(?:age)?\s+([-+\d.eE]+)$/i))) {
      this.set(m[1], 0, Math.min(V_MAX, this.ovp / 1.05), (x) => (this.vSet = x))
      return null
    }
    if ((m = line.match(/^CURR(?:ent)?\s+([-+\d.eE]+)$/i))) {
      this.set(m[1], 0, I_MAX, (x) => (this.iSet = x))
      return null
    }
    if ((m = line.match(/^VOLT(?:age)?:PROT(?:ection)?(?::LEV(?:el)?)?\s+([-+\d.eE]+)$/i))) {
      this.set(m[1], Math.max(OVP_MIN, this.vSet * 1.05), OVP_MAX, (x) => (this.ovp = x))
      return null
    }
    if ((m = line.match(/^CURR(?:ent)?:PROT(?:ection)?:STAT(?:e)?\s+(\S+)$/i)) && BOOL.test(m[1])) {
      this.ocpEnabled = /^(ON|1)$/i.test(m[1])
      return null
    }
    if ((m = line.match(/^OUTP(?:ut)?(?::STAT(?:e)?)?\s+(\S+)$/i)) && BOOL.test(m[1])) {
      const on = /^(ON|1)$/i.test(m[1])
      if (on && this.tripped) this.errors.push(ERR_CONFLICT)
      else this.outputOn = on
      return null
    }
    if (/^OUTP(?:ut)?:PROT(?:ection)?:CLE(?:ar)?$/i.test(line)) {
      if (this.tripped) {
        this.ovTripped = false
        this.ocTripped = false
        this.outputOn = this.outputBeforeTrip
      }
      return null
    }

    this.errors.push(ERR_UNDEF)
    return null
  }
}

import net from 'node:net'
import type { AddressInfo } from 'node:net'
import { N5745aSim } from './n5745a-model'

export { N5745aSim } from './n5745a-model'

export interface N5745aSimHandle {
  port: number
  sim: N5745aSim
  close: () => Promise<void>
}

export function startN5745aSim(port: number, onLog?: (msg: string) => void): Promise<N5745aSimHandle> {
  const sim = new N5745aSim()
  const sockets = new Set<net.Socket>()
  const server = net.createServer((socket) => {
    sockets.add(socket)
    socket.on('close', () => sockets.delete(socket))
    socket.on('error', () => sockets.delete(socket))
    let buf = ''
    socket.on('data', (d) => {
      buf += d.toString('ascii')
      let idx: number
      while ((idx = buf.indexOf('\n')) >= 0) {
        const line = buf.slice(0, idx).replace(/\r$/, '')
        buf = buf.slice(idx + 1)
        onLog?.(`rx: ${line}`)
        const resp = sim.handle(line)
        if (resp !== null) {
          onLog?.(`tx: ${resp}`)
          socket.write(resp + '\n')
        }
      }
    })
  })
  return new Promise((resolve, reject) => {
    server.once('error', reject)
    server.listen(port, () => {
      resolve({
        port: (server.address() as AddressInfo).port,
        sim,
        close: () =>
          new Promise((r) => {
            for (const s of sockets) s.destroy()
            server.close(() => r())
          })
      })
    })
  })
}

import type { ChannelSetpoints, LogSource } from '@shared/types'
import { ScpiConnection } from '../scpi/connection'
import type { PsuDriver } from './types'

const LSR_OVP = 0b0100
const LSR_OCP = 0b1000
// STATus:QUEStionable:CONDition bitleri
const QUES_OV = 1
const QUES_OC = 2

function parseNum(resp: string): number {
  const n = Number(resp.trim().split(',')[0])
  if (!Number.isFinite(n)) throw new Error(`Sayı ayrıştırılamadı: "${resp}"`)
  return n
}

/**
 * Keysight N5700 serisi (ör. N5745A) tek çıkışlı sistem güç kaynağı sürücüsü —
 * LAN raw-socket, TCP 5025, kanal listesiz SCPI.
 *
 * OCP ayrı bir seviye değil, aç/kapat korumadır (CC'ye girince çıkışı keser);
 * arayüz bu türde OCP alanı göstermez, `setOcp` işlem yapmaz. Trip durumu
 * `STAT:QUES:COND?` bitlerinden okunup ortak LSR bit düzenine çevrilir.
 */
export class N5700Driver implements PsuDriver {
  readonly channelCount = 1

  constructor(
    private readonly conn: ScpiConnection,
    private readonly source: LogSource = 'user'
  ) {}

  private check(ch: number): void {
    if (ch !== 1) throw new RangeError(`Geçersiz kanal: ${ch}`)
  }

  async identify(): Promise<string> {
    return this.conn.query('*IDN?', this.source)
  }

  async readChannel(ch: number): Promise<{ voltage: number; current: number; outputOn: boolean }> {
    this.check(ch)
    const [v, i, op] = await Promise.all([
      this.conn.query('MEAS:VOLT?', this.source),
      this.conn.query('MEAS:CURR?', this.source),
      this.conn.query('OUTP?', this.source)
    ])
    return { voltage: parseNum(v), current: parseNum(i), outputOn: op.trim() === '1' }
  }

  async readSetpoints(ch: number): Promise<ChannelSetpoints> {
    this.check(ch)
    const [v, i, ovp] = await Promise.all([
      this.conn.query('VOLT?', this.source),
      this.conn.query('CURR?', this.source),
      this.conn.query('VOLT:PROT?', this.source)
    ])
    return { voltage: parseNum(v), current: parseNum(i), ovp: parseNum(ovp), ocp: 0 }
  }

  async setVoltage(ch: number, volts: number): Promise<void> {
    this.check(ch)
    await this.conn.send(`VOLT ${volts.toFixed(4)}`, this.source)
  }

  async setCurrent(ch: number, amps: number): Promise<void> {
    this.check(ch)
    await this.conn.send(`CURR ${amps.toFixed(4)}`, this.source)
  }

  async setOvp(ch: number, volts: number): Promise<void> {
    this.check(ch)
    await this.conn.send(`VOLT:PROT ${volts.toFixed(4)}`, this.source)
  }

  /** N5700'de OCP seviyesi yok; arayüzde alan da yok — işlem yapılmaz. */
  async setOcp(ch: number, _amps: number): Promise<void> {
    this.check(ch)
  }

  async setOutput(ch: number, on: boolean): Promise<void> {
    this.check(ch)
    await this.conn.send(`OUTP ${on ? 'ON' : 'OFF'}`, this.source)
  }

  async setAllOutputs(on: boolean): Promise<void> {
    await this.conn.send(`OUTP ${on ? 'ON' : 'OFF'}`, this.source)
  }

  async readLsr(ch: number): Promise<number> {
    this.check(ch)
    const cond = parseNum(await this.conn.query('STAT:QUES:COND?', this.source))
    let bits = 0
    if (cond & QUES_OV) bits |= LSR_OVP
    if (cond & QUES_OC) bits |= LSR_OCP
    return bits
  }

  async tripReset(): Promise<void> {
    await this.conn.send('OUTP:PROT:CLE', this.source)
  }

  /** N5700 kanal izleme modu sunmaz — işlevsiz. */
  async setTracking(_on: boolean): Promise<void> {
    // desteklenmiyor (arayüz bu türde tracking göstermez)
  }

  async readTracking(): Promise<boolean> {
    return false
  }

  async readExecutionError(): Promise<number> {
    const resp = await this.conn.query('SYST:ERR?', this.source)
    // "-222,\"Data out of range\"" → mutlak kod (DeviceManager > 0 kontrolü için)
    return Math.abs(parseNum(resp))
  }
}
```
